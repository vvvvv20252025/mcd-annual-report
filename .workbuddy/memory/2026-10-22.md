  function fmtHour(min){ min = Math.round(min); var h = Math.floor(min/60)%24; var m = min%60; return (h<10?"0":"")+h+":"+(m<10?"0":"")+m; }
  function fmtDate(o){ return o.year + "." + (o.month<10?"0":"") + o.month + "." + (o.day<10?"0":"") + o.day; }
  function buildDist(orders){
    var buckets = {morning:0, lunch:0, tea:0, dinner:0, night:0};
    orders.forEach(function(o){
      if (o.hour==null) { buckets.lunch += o.count; return; }
      if (o.hour < 11) buckets.morning += o.count;
      else if (o.hour < 14) buckets.lunch += o.count;
      else if (o.hour < 17) buckets.tea += o.count;
      else if (o.hour < 22) buckets.dinner += o.count;
      else buckets.night += o.count;
    });
    var s = orders.reduce(function(a,o){return a+o.count;},0) || 1;
    return [
      Math.round(buckets.morning/s*100), Math.round(buckets.lunch/s*100),
      Math.round(buckets.tea/s*100), Math.round(buckets.dinner/s*100), Math.round(buckets.night/s*100)
    ];
  }

  function parseDate(s){
    // 支持 2025-06-20 23:30 / 2025-06-20 / 2025/6/20 23:30
    var m = s.match(/(\d{4})[/-](\d{1,2})[/-](\d{1,2})(?:[ T](\d{1,2}):(\d{2}))?/);
    if (!m) return { year:0 };
    return { year:+m[1], month:+m[2], day:+m[3], hour: m[4]!=null?+m[4]:null, min: m[5]!=null?+m[5]:0,
      ts: new Date(+m[1], +m[2]-1, +m[3], m[4]!=null?+m[4]:12, m[5]!=null?+m[5]:0).getTime() };
  }

### 开发中修复的真实缺陷（均已复验）
1. **积分全读 0** —— `query-my-account` 数据在 `data` 嵌套层，pick 未向下穿透 → 加 data/result/account/info 穿透。
2. **深夜均时算出 12:32** —— 22:47 与 00:58 直接算术平均跨零点 → 改为「以 20:00 为原点平移再折回」。
3. **订单数与件数口径混用** —— 一单多品导致 total=303 → 拆为 `total`(下单次数) 与 `itemCount`(件数)，Top5 按件、曲线与深夜占比按单。
4. **模板 JS 语法错误 `missing ) after argument list`** —— 用正则删交互代码块截断了 `var ptTotal = parseFloat(` → 改为**不删代码 + bind() 判空辅助函数**。
5. **`render is not defined`** —— 交互版整个 JS 包在 `(function(){"use strict";...})()` 里，render 是 IIFE 内部函数 → 改为**把渲染调用注入 IIFE 内部、紧跟 "use strict"; 之后**（函数声明已提升）。
6. **`.reveal` 内容不可见** —— 静态报告无滚动触发 → 注入时直接给所有 `.reveal` 加 `in` 类并把 count-up 数字设为终值。
7. **appendChild null / IO observe null 报错** —— orderRows 给空壳对象；`io.observe(x)` 换成 `observe(io,x)` 判空辅助。
8. **GIF 编码 NaN** —— `quantize` 需 `new Uint8Array(png.data)` 且用 256 色；过渡帧的 `window.scrollTo` 在 scroll-snap 下回弹导致帧尺寸异常 → 改用 transform 位移制造滚动观感。
9. **draw.io PNG 空白** —— ① 所有 mxCell 必须在 `<root>` 内；② cell id 必须唯一（改显式 id）；③ style 里 fontFamily 含逗号必须 `&quot;` 包裹；④ **value 内嵌 HTML 富文本导致节点被丢弃 → 改纯文本 + `&#10;` 换行**（PNG 从 88KB 跳到 438KB）。
